function getCropProblems(){

.getElementById(
'cropSelect'
)
.value;

DATA.Crop_Problem

.filter(

function(link){

link.Crop_ID===

cropID

)

.map(

function(link){

link.Problem_ID

}

);

.filter(

function(problem){

ids.includes(

problem.Problem_ID

)

);


}

);


}

/* =========================================
   PROBLEM CHANGE
========================================= */

function problemChanged(
boxID
){


const box=

document
.getElementById(
boxID
);


const problemID=

box
.querySelector(
'.problem-select'
)
.value;


const problem=

DATA.Problems

.find(

function(item){


return(

item.Problem_ID===

problemID

);


}

);


const info=

box.querySelector(
'.info-box'
);


if(!problem){


info.classList.add(
'hidden'
);


return;


}


info.classList.remove(
'hidden'
);


const image=

box.querySelector(
'.info-image'
);


image.style.display=
'block';


image.src=

problem.Image_URL||'';


image.onerror=

function(){


this.style.display=
'none';


};


box
.querySelector(
'.cause'
)
.textContent=

problem.Cause_Type||'';


box
.querySelector(
'.symptoms'
)
.textContent=

problem.Symptoms_Damage||'';


const aiSelect=

box.querySelector(
'.ai-select'
);


aiSelect.innerHTML=

`

<option value="">

-- Active Ingredient নির্বাচন করুন --

</option>

`;


DATA.Problem_AI

.filter(

function(link){


return(

link.Problem_ID===

problemID

);


}

)

.forEach(

function(link){


const ai=

DATA.Active_Ingredients

.find(

function(item){


return(

item.AI_ID===

link.AI_ID

);


}

);


if(!ai){

return;

}


const option=

document
.createElement(
'option'
);


option.value=

ai.AI_ID;


option.textContent=

ai.Active_Ingredient+

(

ai.Formulation

?

' ('+

ai.Formulation+

')'

:

''

);


aiSelect.appendChild(
option
);


}

);


loadManagement(

box,

problemID

);


}

/* =========================================
   ESCAPE
========================================= */

function escapeHTML(
value
){


return String(

value||

''

)

.replace(

/&/g,

'&amp;'

)

.replace(

/</g,

'&lt;'

)

.replace(

/>/g,

'&gt;'

)

.replace(

/"/g,

'&quot;'

);


}
